name: Release

on:
  workflow_dispatch:
  push:
    tags:
      - "v*"

permissions:
  contents: write

jobs:
  build-windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - uses: dtolnay/rust-toolchain@stable
      - uses: Swatinem/rust-cache@v2
        with:
          workspaces: src-tauri
      - run: npm ci
      - name: Build and upload Windows installers
        uses: tauri-apps/tauri-action@v0
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        with:
          tagName: v__VERSION__
          releaseName: "AI Usage Dock v__VERSION__"
          releaseBody: "See the assets below to download and install AI Usage Dock."
          releaseDraft: true
          prerelease: false
          args: --bundles nsis,msi

  # Linux and macOS builds are intentionally left commented until the
  # platform-specific tray/window behavior has been verified.
  #
  # build-linux:
  #   runs-on: ubuntu-24.04
  #   ...
  #
  # build-macos:
  #   runs-on: macos-latest
  #   ...
